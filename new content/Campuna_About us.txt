4. Schema code (JSON-LD, copy-paste ready)

Paste each block into the <head>. Both are already embedded in the HTML file. The Organization block itself lives on the homepage; this page references it by @id.


4.1 BreadcrumbList

<script type="application/ld+json">

{

  "@context": "https://schema.org",

  "@type": "BreadcrumbList",

  "itemListElement": [

    { "@type": "ListItem", "position": 1, "name": "Startseite", "item": "https://campuna.de/" },

    { "@type": "ListItem", "position": 2, "name": "Über Campuna", "item": "https://campuna.de/ueber-uns" }

  ]

}

</script>

 


4.2 AboutPage

<script type="application/ld+json">

{

  "@context": "https://schema.org",

  "@type": "AboutPage",

  "@id": "https://campuna.de/ueber-uns#aboutpage",

  "url": "https://campuna.de/ueber-uns",

  "name": "Über Campuna",

  "inLanguage": "de-DE",

  "isPartOf": { "@id": "https://campuna.de/#website" },

  "about": { "@id": "https://campuna.de/#organization" },

  "description": "Warum es Campuna gibt, wofür wir stehen und wie unser Camping-Marktplatz für Deutschland wächst."

}

</script>
